namespace ReelDiscovery.Services.Case;
public static class CaseFolderNaming {
  public const int MaxLength = 80;
  public const string Untitled = "Untitled Case";
  public static string CaseTitle(CaseDraft draft);        // "Brightline Glazing, LLC v. Harmon Builders, Inc." / memo "{Memo.Organization or first party Name} - {Memo.Re}" / doc.Title / Untitled
  public static string DefaultFolderName(CaseDraft draft); // pleadings use ShortNames "Brightline v. Harmon"; memo = CaseTitle; then Sanitize
  public static string Sanitize(string? name);
  public static string UniqueFolderPath(string root, string name);  // root\name, root\name (2), ...; throws InvalidOperationException if the full path escapes root
}
public enum CaseLoadError { None, Missing, TooLarge, NewerVersion, Corrupt }
public class CaseLoadResult { public CaseDraft? Draft; public CaseLoadError Error; public string Message = ""; public bool Succeeded => Draft != null; }
public class RecentCase { public string Path = ""; public string Title = ""; public CaseStage Stage; public DateTime LastOpenedUtc; }
public class CaseStore {
  public const string CaseFileName = "case.json";
  public const long MaxCaseFileBytes = 50L * 1024 * 1024;
  public const int MaxRecentCases = 8;
  public CaseStore(string? casesRoot = null, string? appDataFolder = null);
  public string CasesRoot { get; }                    // default MyDocuments\ReelDiscovery\Cases
  public DateTime? LastSavedUtc { get; }
  public string LastSaveError { get; }                // empty after a successful save
  public event EventHandler? SaveCompleted;           // raised after every save attempt (success or failure)
  public static bool CanSave(CaseDraft? draft);        // draft?.CoreDocument != null
  public string EnsureFolder(CaseDraft draft);         // creates the folder on first call; sets draft.FolderPath; returns it
  public Task<bool> SaveAsync(CaseDraft draft, CancellationToken ct = default);
  public bool SaveNow(CaseDraft draft);               // synchronous flush for closing
  public CaseLoadResult Load(string caseJsonPath);
  public List<RecentCase> GetRecentCases();
  public void TouchRecent(CaseDraft draft);           // upsert {FolderPath\case.json, CaseTitle, Stage, now}, cap 8
  public void RemoveRecent(string caseJsonPath);
  public string MoveFolder(CaseDraft draft, string targetParent);   // returns new folder; throws IOException/UnauthorizedAccessException with the OS reason
  public static Task WriteAtomicAsync(string path, byte[] bytes, CancellationToken ct = default);
  public static void WriteAtomic(string path, byte[] bytes);
}
// Models/WizardState.cs additions
public CaseStore CaseStore { get; set; } = new();
/// One-shot status text the next case step shows on entry (e.g. "Opened {title}."), then clears.
public string PendingNotice { get; set; } = string.Empty;

```csharp
namespace ReelDiscovery.UserControls;
public interface IGuideHost {
  CaseDraft Draft { get; }
  IReadOnlyList<GuideFinding> Findings { get; }
  bool IsBusy { get; }
  string WritingEntityId { get; }                        // "" when none
  void NotifyEdited(GuidePart part);                     // revalidate, autosave, refresh captions/banner/checks/dots
  void SetStatus(string text, Color color);
  Task RegenerateEntryAsync(GuidePart part, string entityId, string guidance);   // host runs the gateway call + status
  Task RegenerateTabAsync(GuidePart part);               // host shows D1 and runs
  Task RetryIssueDetailsAsync();
  void ShowTab(GuidePart part, string? entityId = null);
}
public abstract class GuideTabPanel : UserControl {
  protected GuideTabPanel(GuidePart part, string caption, string noun, bool hasToolbar, string guidancePlaceholder);
  public GuidePart Part { get; }
  public void Attach(IGuideHost host);
  protected IGuideHost Host { get; }
  public abstract IReadOnlyList<Control> TabActions { get; }      // buttons the shell shows on the tabs row
  public void RefreshFromDraft();                                  // rebuild rows keeping sort/selection, refresh detail + header count
  public virtual bool SelectEntity(string id);                     // clears a hiding filter, selects, scrolls into view, focuses first field
  public virtual void ShowNeedsReview() { }
  public virtual void CommitPendingEdits() { }
  public virtual void SetBusy(bool busy);                          // grid ReadOnly, fields ReadOnly, toolbar disabled
  // protected building blocks for subclasses:
  protected DataGridView Grid; protected Label Caption; protected FlowLayoutPanel HeaderRight; protected Label CountLabel;
  protected Label DetailHeading; protected Label DetailMeta; protected Panel NoticeSlot; protected TableLayoutPanel Fields;
  protected TextBox GuidanceBox; protected ThemedButton DeleteButton; protected ThemedButton RegenerateButton;
  protected string? SelectedId { get; }
  protected abstract void BuildColumns();
  protected abstract IEnumerable<(string Id, object?[] Cells)> BuildRows();
  protected abstract void ShowDetail(string? id);                  // fill Fields for id (null -> empty detail state)
  protected virtual Color DotColor(string id);                     // default: Danger / Warning / Success from Host.Findings; Accent when id == Host.WritingEntityId
  protected Label AddCaptionedLabel(string caption, string text);  // read-only field
  protected CardPanel AddTextField(string caption, TextBox box, int lines = 1, string? hint = null);
  protected (CardPanel Left, CardPanel Right) AddPairedTextFields(string captionLeft, TextBox left, string captionRight, TextBox right);
  protected ComboBox AddCombo(string caption, IEnumerable<string> items);
  protected void ShowFieldError(Control field, string message);   // Danger border + SmallFont Danger line under the field
  protected void ClearFieldErrors();
  protected void ClearFields();
}
public class LinkItem { public string Id = ""; public string Text = ""; }
public class LinkPicker : UserControl {
  public LinkPicker(string filterPlaceholder);
  public void SetItems(IReadOnlyList<LinkItem> items, IEnumerable<string> checkedIds, IEnumerable<string>? lockedIds = null, string lockedSuffix = " (did it)");
  public IReadOnlyList<string> CheckedIds { get; }
  public bool ReadOnly { get; set; }
  public event EventHandler? CheckedChanged;
}
public class IssuesTabPanel : GuideTabPanel { public IssuesTabPanel(); }
```

namespace ReelDiscovery.Services.Ai;

public interface IBatchTransport
{
    LlmProviderKind Kind { get; }
    int MaxRequestsPerBatch { get; }          // 10_000 for both providers (packaging cap, discretion)
    long MaxBatchBytes { get; }               // 150 * 1024 * 1024
    long? MaxEstimatedInputTokens { get; }    // OpenAI 2_000_000 (Tier-1 queue limit, Pitfall 6); Anthropic null
    Task<BatchSubmitResult> SubmitAsync(IReadOnlyList<BatchItemRequest> items, CancellationToken ct = default);
    Task<BatchStatusResult> GetStatusAsync(string batchId, CancellationToken ct = default);
    Task<BatchResultsResult> GetResultsAsync(string batchId, CancellationToken ct = default);
    Task CancelAsync(string batchId, CancellationToken ct = default);                        // best effort; only OCE escapes
    Task DeleteFilesAsync(string? inputFileId, string? outputFileId, CancellationToken ct = default); // OpenAI; no-op Anthropic
}
public class BatchItemRequest { public string CustomId { get; set; } = ""; public LlmProviderRequest Request { get; set; } = new(); }
public class BatchSubmitResult { public bool Succeeded; public string BatchId = ""; public string InputFileId = ""; public LlmFailureKind FailureKind; public string Message = ""; } // as auto-properties
public enum BatchState { InProgress, Ended, Failed, Expired, Cancelled }
public class BatchStatusResult { bool Succeeded; BatchState State; int Done; int Total; int Failed; string OutputFileId; string ErrorFileId; LlmFailureKind FailureKind; string Message; }
public enum BatchItemOutcome { Succeeded, Errored, Expired, Canceled }
public class BatchItemResult { string CustomId; BatchItemOutcome Outcome; LlmProviderResponse? Response; string Message; }
public class BatchResultsResult { bool Succeeded; List<BatchItemResult> Items; LlmFailureKind FailureKind; string Message; }

public enum BatchLedgerRecordType { Submitted, Collected, Cancelled, OrphansBilled, Consumed }   // append-only
public class BatchLedgerItem { string CustomId; string Operation; string ModelId; }   // ModelId prices orphans
public class BatchLedgerRecord { BatchLedgerRecordType Type; DateTime Utc; LlmProviderKind Provider; string BatchId; Guid PlanId; DateTime SubmittedUtc; string StageLabel; string InputFileId; string OutputFileId; List<BatchLedgerItem> Items; int Done; int Total; }
public class BatchLedgerState { Dictionary<string, BatchLedgerRecord> SubmittedById; HashSet<string> CollectedIds; HashSet<string> CancelledIds; HashSet<string> OrphansBilledIds; HashSet<string> ConsumedCustomIds; Dictionary<string, string> BatchIdByCustomId; IEnumerable<BatchLedgerRecord> Pending(Guid planId); }
public class BatchLedger
{
    public const string FileName = "provider-batches.jsonl";
    public const string ResultsFolderName = "batch-results";
    public BatchLedger(string caseFolder);
    public static string PathFor(string caseFolder);
    public Task AppendAsync(BatchLedgerRecord record, CancellationToken ct = default);
    public BatchLedgerState Replay();                                   // never throws
    public void SaveResults(string batchId, IReadOnlyList<BatchItemResult> items);   // atomic; throws IOException on failure
    public IReadOnlyList<BatchItemResult> ReadResults(string batchId);  // never throws; empty when missing
    public string ResultsPathFor(string batchId);
    public static string SafeFileId(string providerId);
    public static int PendingBatchCount(string caseFolder, Guid planId); // Submitted without Collected/Cancelled
}
public static class RequestIdentity { public static string CustomIdFor(LlmProviderKind kind, LlmProviderRequest request); }

public enum LlmTier { Story, Volume }
public enum LlmProviderKind { OpenAI, Anthropic }

public class LlmRequest
{
    public LlmTier Tier { get; set; }
    public string OperationName { get; set; } = string.Empty;   // "Storyline Generation"
    public string SystemPrompt { get; set; } = string.Empty;    // stable prefix (byte-identical across calls)
    public string UserPrompt { get; set; } = string.Empty;      // variable suffix
    public int MaxOutputTokens { get; set; } = 16_000;          // includes thinking/reasoning tokens
}

public enum LlmOutcome { Succeeded, Refused, Truncated, Invalid, Failed }
public enum LlmFailureKind { None, Auth, SpendLimit, RateLimited, Transient, BadRequest, Network, Unknown }

public class LlmResult
{
    public LlmOutcome Outcome { get; set; }
    public LlmFailureKind FailureKind { get; set; }
    public string? RawText { get; set; }
    public string? Message { get; set; }          // refusal text / error message; never contains an API key
    public string OperationName { get; set; } = string.Empty;
    public LlmProviderKind Provider { get; set; }
    public string ModelId { get; set; } = string.Empty;
    public int Attempts { get; set; }
    public bool IsFatal => FailureKind is LlmFailureKind.Auth or LlmFailureKind.SpendLimit;
}

public class LlmResult<T> : LlmResult where T : class
{
    public T? Value { get; set; }
    public T GetValueOrThrow();   // Succeeded && Value != null ? Value : throw new LlmCallException(this)
}

public class LlmCallException : Exception
{
    public LlmCallException(LlmResult result);   // Message: "{OperationName}: {Outcome}[ ({FailureKind})] - {Message}"
    public LlmResult Result { get; }
}

public class LlmKeyTestResult
{
    public bool Succeeded { get; set; }
    public LlmFailureKind FailureKind { get; set; }
    public string Message { get; set; } = string.Empty;   // user-facing status text; never contains the key
}

public interface ILlmGateway
{
    LlmProviderKind Provider { get; }
    AIModelConfig ResolveModel(LlmTier tier);
    Task<LlmResult<T>> GenerateAsync<T>(LlmRequest request, CancellationToken ct = default) where T : class;
}

public class LlmProviderRequest
{
    public string ModelId { get; set; } = string.Empty;
    public string? Effort { get; set; }                 // sent only when non-null
    public string SchemaName { get; set; } = string.Empty;
    public string SchemaJson { get; set; } = string.Empty;
    public string SystemPrompt { get; set; } = string.Empty;
    public string UserPrompt { get; set; } = string.Empty;
    public int MaxTokens { get; set; }
}

public class LlmProviderResponse
{
    public LlmOutcome Outcome { get; set; }             // Succeeded here means "stop reason OK"; the gateway still parses
    public LlmFailureKind FailureKind { get; set; }
    public string? Text { get; set; }
    public string? Message { get; set; }
    public int InputTokens { get; set; }
    public int OutputTokens { get; set; }
}

public interface ILlmProvider
{
    LlmProviderKind Kind { get; }
    // Never throws for API/HTTP failures (maps them to Outcome=Failed + FailureKind); rethrows OperationCanceledException.
    Task<LlmProviderResponse> SendAsync(LlmProviderRequest request, CancellationToken ct = default);
    // Free key check (model retrieve); never throws except OperationCanceledException.
    Task<LlmKeyTestResult> TestKeyAsync(string modelId, CancellationToken ct = default);
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Provider response -> app | Model text is untrusted; only typed DTO deserialization crosses it |
| App -> paid API | Each retry spends money |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-09 | Denial of wallet | LlmGateway retry loop | mitigate | Refused/Auth/SpendLimit/RateLimited/Transient/BadRequest get 0 app retries; Invalid capped at 3 attempts; Truncated capped at 1 retry and model ceiling; tested by LlmGatewayTests |
| T-01-10 | Tampering | Hostile model JSON | mitigate | Strict schema (additionalProperties:false) + typed deserialization only; parse failures classified Invalid, never executed |

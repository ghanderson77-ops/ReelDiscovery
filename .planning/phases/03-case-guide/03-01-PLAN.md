namespace ReelDiscovery.Models.Case;
public enum CaseStage { Described, CoreDocument, CaseGuide, Summary }          // CaseStage.cs
public enum GuidePart { Issues, Cast, Facts, Values }                            // GuidePartState.cs
public enum GuidePartStatus { Pending, Generating, Ready, Failed }               // GuidePartState.cs

public enum OrganizationKind { Party, LawFirm, Agency, ThirdParty, Individual }
public class CaseOrganization { string Id; string Name; string ShortName; OrganizationKind Kind; string PartyId; List<string> RepresentsOrgIds; string Domain; string Location; string TimeZoneId; }
public enum CastTier { Key, Background }
public enum CastRole { Party, InHouseCounsel, OutsideCounsel, Staff }
public enum SignatureStyle { Full, Short, NameOnly, MobileFooter }
public class CastMember { string Id; string FirstName; string LastName; [JsonIgnore] string FullName; string Title; string Department; string OrganizationId; CastTier Tier; CastRole Role; string Email; bool EmailIsUserEdited; string Phone; string ActiveFrom; string ActiveTo; string Involvement; string VoiceNote; SignatureStyle SignatureStyle; string Signature; bool DetailsPending; bool IsUserEdited; }
public enum FactSignificance { Hot, Key, Supporting }
public class KeyFact { string Id; string Date; string Title; string Description; List<string> IssueIds; List<string> ActorIds; List<string> KnownByIds; FactSignificance Significance; string SourceSectionId; List<string> MappingNotes; bool IsUserEdited; bool NeedsReview; string ReviewReason; }
public enum ValueKind { Amount, InvoiceNumber, ChangeOrderNumber, ContractNumber, PurchaseOrder, DocumentTitle, ProjectName, AccountNumber, Address, Other }
public class CanonicalValue { string Id; ValueKind Kind; string Label; string Value; string Description; List<string> FactIds; List<string> MappingNotes; bool IsUserEdited; bool NeedsReview; string ReviewReason; }
public class CaseGuide {
  List<CaseIssue> Issues; List<CaseOrganization> Organizations; List<CastMember> Cast; List<KeyFact> Facts; List<CanonicalValue> Values;
  Dictionary<string,int> IdCounters;
  GuidePartState IssuesState, IssueDetailsState, CastState, FactsState, ValuesState;
  string DismissedStaleKey;
  GuidePartState State(GuidePart part);
  CaseIssue? FindIssue(string? id); CaseOrganization? FindOrganization(string? id); CastMember? FindCast(string? id); KeyFact? FindFact(string? id); CanonicalValue? FindValue(string? id);
  IEnumerable<string> AllIds();
  string ComputeHash();
  void Normalize();
}

public CaseGuide? Guide { get; set; }

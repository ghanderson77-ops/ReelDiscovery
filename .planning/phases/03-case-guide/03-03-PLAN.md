namespace ReelDiscovery.Services.Case;
public static class IdAllocator {
  public const string Issue = "ISS", Organization = "ORG", Character = "CH", Fact = "F", Value = "V";
  public static int WidthFor(string prefix);                   // ISS/ORG -> 2, others -> 3
  public static string Next(CaseGuide guide, string prefix);   // max(existing ids, counter) + 1; stores counter; zero-padded
  public static string? Normalize(string? raw, string prefix); // " ch-7 " -> "CH-007"; "CH-0007" -> "CH-007"; "Michael" -> null; other prefix -> null
}
public static class IssueExtractor {
  public static List<CaseIssue> Extract(CoreDocument doc, CaseGuide guide);
  public static string SourceLabel(CoreDocument doc, CaseIssue issue);   // "Count II" (pleadings) / "Finding 2" (memo)
  internal static string NormalizeTitle(string? title);
}
public static partial class EmailIdentity {
  public static string BaseName(CaseOrganization org);
  public static string DomainFor(CaseOrganization org, ISet<string> takenDomains);
  public static string AddressFor(string firstName, string lastName, string domain, ISet<string> takenAddresses);
  public static string Fold(string text, bool keepHyphen = false);
  public static bool IsReservedDomain(string? domainOrEmail);
  public static bool IsWellFormed(string? email);
  public static string DomainOf(string? email);
}
public static class SignatureBuilder {
  public static string Build(CastMember member, CaseOrganization? org);
  public static string FictionalPhone();
}
public static class UsTimeZones {
  public static string ForLocation(string? location);   // "Chicago, Illinois" -> "America/Chicago"; unknown -> "America/New_York"
}

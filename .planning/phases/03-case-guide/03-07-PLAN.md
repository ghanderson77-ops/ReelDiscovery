public static class GuideStaleness {
  public static readonly GuidePart[] Order = { GuidePart.Issues, GuidePart.Cast, GuidePart.Facts, GuidePart.Values };

  public static string BasisFor(CaseDraft draft, GuidePart part);    // Issues/Cast <- core; Facts <- core + issues(Id,Title) + cast(Id,FullName,OrganizationId,ActiveFrom,ActiveTo); Values <- core + facts(Id,Date,Title,Description)
  public static void Stamp(CaseDraft draft, GuidePart part);         // BasisHash = BasisFor, GeneratedUtc = UtcNow
  public static bool IsStale(CaseDraft draft, GuidePart part);       // Status Ready && BasisHash != "" && BasisHash != BasisFor
  public static List<GuidePart> StaleParts(CaseDraft draft);         // in Order
  public static string StaleKey(CaseDraft draft);                    // "" when nothing stale; else joined stale parts + their current bases (for Not Now)
  public static void AbsorbHandEdit(CaseDraft draft, GuidePart editedPart, Action edit);  // parts after editedPart that were fresh stay fresh (restamped)
  public static int NeedsReviewCount(CaseDraft draft);
}

  public static GuideEditResult SetFactLinks(CaseDraft draft, string factId, FactLinkKind kind, IReadOnlyList<string> ids);
  public static bool MarkReviewed(CaseDraft draft, string factId);
  public static int ValuesReferencing(CaseDraft draft, string factId);
  public static bool DeleteFact(CaseDraft draft, string factId);
  public static CanonicalValue AddValue(CaseDraft draft);
  public static GuideEditResult SetValueField(CaseDraft draft, string valueId, ValueField field, string value);

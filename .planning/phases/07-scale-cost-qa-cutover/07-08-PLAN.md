public static class BatchStage
{
    // AsyncLocal flows into every task the runner starts inside a using block
    public static string Current { get; }            // "batch" when unset

public enum BatchStatusKind { Submitted, Progress, Finished, FinishedWithFailures, FailedAtProvider, CollectedAfterRestart }
public class BatchStatusEvent { BatchStatusKind Kind; string BatchId; string StageLabel; LlmProviderKind Provider; DateTime SubmittedUtc; int Done; int Total; int Succeeded; int Failed; string Reason; DateTime NextCheckUtc; TimeSpan Duration; }
public class BatchingLlmProvider : ILlmProvider
{
    public BatchingLlmProvider(ILlmProvider live, IBatchTransport transport, BatchLedger ledger, Guid planId,
        Func<string, AIModelConfig?> priceLookup, BatchTimings? timings = null);
    public event Action<BatchStatusEvent>? StatusChanged;   // raised off the UI thread
    public int PendingBatchCount { get; }

    public void Detach();                                   // stop polling; provider batches keep running
    public Task<(CostBreakdown Cost, int InputTokens, int OutputTokens)> BillOrphansAsync(CancellationToken ct = default);
    public static Task CancelPendingAsync(BatchLedger ledger, Func<LlmProviderKind, IBatchTransport?> transportFor, Guid planId, CancellationToken ct = default); // Start Over; one transport per ledger record's provider
